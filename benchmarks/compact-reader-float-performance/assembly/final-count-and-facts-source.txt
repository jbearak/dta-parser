1: //! Allocation-specific facts validated during eligible Arrow FLOAT scans.
2: //! No generic descriptor factory grants these facts. Coverage is deliberately
3: //! stricter than the missing-count cache: incomplete or reordered spans decline.
4: 
5: use crate::{FormatVersion, NumericData, NumericKind, TemporalKind};
6: 
7: const STRICT_MODERN_FLOAT: u32 = 1;
8: const FLOAT_BOUNDS_KNOWN: u32 = 2;
9: const ZERO_COUNT_KNOWN: u32 = 4;
10: 
11: #[inline(always)]
12: pub(crate) fn is_canonical_modern_float_tag(bits: u32) -> bool {
13:     let offset = bits.wrapping_sub(0x7f00_0000);
14:     (offset <= 26 * 2048) & (offset & 2047 == 0)
15: }
16: 
17: #[inline(always)]
18: pub(crate) fn modern_float_missing(bits: u32) -> bool {
19:     is_canonical_modern_float_tag(bits) | ((bits & 0x7fff_ffff) > 0x7f80_0000)
20: }
21: 
22: #[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
23: pub(crate) struct NumericFacts {
24:     flags: u32,
25:     maximum: u32,
26:     minimum_nonzero: u32,
27:     zeros: usize,
28: }
29: 
30: impl NumericFacts {
31:     pub(crate) fn publish(self, descriptor: &mut NumericData) {
32:         descriptor.domain_flags = self.flags;
33:         descriptor.float_max_magnitude_bound = self.maximum;
34:         descriptor.float_min_nonzero_magnitude_bound = self.minimum_nonzero;
35:         descriptor.zero_count = self.zeros;
36:     }
37: }
38: 
39: #[derive(Clone, Copy, Debug)]
40: pub(crate) struct FloatFacts {
41:     valid: bool,
42:     maximum: u32,
43:     minimum_nonzero: u32,
44:     zeros: usize,
45: }
46: 
47: impl FloatFacts {
48:     pub(crate) fn new() -> Self {
49:         Self {
50:             valid: true,
51:             maximum: 0,
52:             minimum_nonzero: u32::MAX,
53:             zeros: 0,
54:         }
55:     }
56: 
57:     #[inline(always)]
58:     pub(crate) fn observe(&mut self, bits: u32) -> bool {
59:         let magnitude = bits & 0x7fff_ffff;
60:         let observed = magnitude <= 0x7eff_ffff;
61:         let canonical_tag = is_canonical_modern_float_tag(bits);
62:         self.valid &= observed | canonical_tag;
63:         self.maximum = self.maximum.max(if observed { magnitude } else { 0 });
64:         self.minimum_nonzero = self.minimum_nonzero.min(if observed && magnitude != 0 {
65:             magnitude
66:         } else {
67:             u32::MAX
68:         });
69:         self.zeros += usize::from(observed && magnitude == 0);
70:         // Modern missing counts include exactly the 27 canonical positive
71:         // tags or either-sign IEEE NaNs. Neither infinity nor noncanonical
72:         // high finite codes are missing, even though they decline the proof.
73:         canonical_tag | (magnitude > 0x7f80_0000)
74:     }
